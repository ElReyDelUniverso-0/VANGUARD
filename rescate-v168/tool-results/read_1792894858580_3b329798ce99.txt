    33→function leerMente(): { recompensados: string[]; dia: string; n: number; chatDia: string; chatHecho: boolean } {
    34→  try {
    35→    const raw = JSON.parse(localStorage.getItem(LS_MENTE) || "{}");
    36→    const dia = new Date().toISOString().slice(0, 10);
    37→    return {
    38→      recompensados: Array.isArray(raw.recompensados) ? raw.recompensados : [],
    39→      dia: typeof raw.dia === "string" ? raw.dia : "",
    40→      n: typeof raw.n === "number" ? raw.n : 0,
    41→      chatDia: typeof raw.chatDia === "string" ? raw.chatDia : "",
    42→      chatHecho: raw.chatHecho === true,
    43→    };
    44→  } catch {
    45→    return { recompensados: [], dia: "", n: 0, chatDia: "", chatHecho: false };
    46→  }

   233→        role: "mente",
   234→        text: respuesta,

   238→        claves: Array.isArray(data.claves) ? data.claves : undefined,
   239→        confianza: typeof data.confianza === "number" ? data.confianza : undefined,
   240→      }];
   241→      setChat(conIa);
   242→      localStorage.setItem(LS_CHAT, JSON.stringify(conIa.slice(-24)));
   243→    } catch {
   244→      toast.error("LA MENTE no responde — el pulso sigue de todos modos");
   245→    } finally {
   246→      setPensando(false);
   247→    }
