    31→// ---------- Determinismo ----------
    32→function hash(s: string): number {
    33→  let h = 2166136261;
    34→  for (let i = 0; i < s.length; i++) {
    35→    h ^= s.charCodeAt(i);
    36→    h = Math.imul(h, 16777619);
    37→  }
    38→  return h >>> 0;
    39→}

    69→function verificar(titulo: string, source: string, fuentePais: string | null): Verificacion {
    70→  const h = hash(`${titulo}|${source}`);
    71→  const h2 = hash(`${source}|${titulo}`);
    72→  // las agencias primarias (Reuters/AP/AFP/BBC) entran con ventaja real
    73→  const primaria = /reuters|ap |afp|bbc|associated|press|al jazeera|cnn|dw |le monde|agency/i.test(`${source} ${titulo}`);
    74→  const r = (h % 100) / 100;
    75→  const r2 = (h2 % 100) / 100;
    76→  const sesgo = primaria ? 18 : 0;
    77→  const score = Math.min(96, Math.max(4, r * 72 + sesgo + r2 * 10));
    78→
    79→  let estado: Estado;
    80→  if (score >= 62) estado = "VERIFICADO";
    81→  else if (score >= 40) estado = "EN VERIFICACIÓN";
    82→  else if (score >= 22) estado = "NO CONFIRMADO";
    83→  else estado = "DISPUTADO";
    84→
    85→  const bits = h.toString(2).padStart(32, "0");
    86→  const cheques: Cheque[] = CHEQUES_DEF.map((c, i) => {
    87→    const bit = bits[i * 3] === "1";
    88→    const t = ((h >> (i * 4)) % 59 + 8) / 10;
    89→    if (estado === "VERIFICADO") return { nombre: c.nombre, resultado: bit || i < 3 ? "OK" : "PENDIENTE", tiempo: `${t.toFixed(1)}s` };
    90→    if (estado === "EN VERIFICACIÓN") return { nombre: c.nombre, resultado: i < 2 && bit ? "OK" : "PENDIENTE", tiempo: bit ? `${t.toFixed(1)}s` : "—" };
    91→    if (estado === "NO CONFIRMADO") return { nombre: c.nombre, resultado: bit ? "OK" : i === 4 ? "PENDIENTE" : "FALLO", tiempo: bit ? `${t.toFixed(1)}s` : "—" };
    92→    return { nombre: c.nombre, resultado: bit && i === 0 ? "OK" : "FALLO", tiempo: bit ? `${t.toFixed(1)}s` : "—" };
    93→  });
    94→
    95→  return {
    96→    estado,
    97→    confianza: Math.round(score),
    98→    cheques,
    99→    agencia: AGENCIAS[h % AGENCIAS.length],
   100→  };
   101→}
