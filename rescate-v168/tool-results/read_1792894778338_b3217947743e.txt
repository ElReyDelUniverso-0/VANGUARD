    29→interface BkStats { staked: number; payout: number; bets: number; wins: number; }
    30→function loadStats(): BkStats {
    31→  try { return JSON.parse(localStorage.getItem("vanguard-bk-stats") ?? "") as BkStats; }
    32→  catch { return { staked: 0, payout: 0, bets: 0, wins: 0 }; }
    33→}

    35→function countdown(ms: number): string {
    36→  if (ms <= 0) return "EN JUEGO";
    37→  const s = Math.floor(ms / 1000);
    38→  return `${String(Math.floor(s / 60)).padStart(2, "0")}:${String(s % 60).padStart(2, "0")}`;
    39→}

    65→  useEffect(() => { jackpotRef.current = jackpot; }, [jackpot]);

    81→    if (refunded.length > 0) {
    82→      for (const t of tks) {
    83→        if (t.status === "ABIERTA" && t.legs.some((l) => !ids.has(l.eventId))) {
    84→          t.status = "COBRADA"; t.payout = t.stake;
    85→        }

    89→      toast.info(`Jornada rotada: ${refunded.length} apuesta(s) reembolsada(s)`, { description: `+${total} mon de vuelta` });
    90→      saveTickets(tks);
    91→    }

    93→    setEvents([...evs]);
    94→    ticketsRef.current = tks;

   102→      tick.current++;
   103→      const evs = eventsRef.current;
   104→      if (!evs.length) return;
   105→      const next = evs.map((e) => {
   106→        const c = { ...e, corners: [...e.corners] as [number, number], cards: [...e.cards] as [number, number], xg: [...e.xg] as [number, number] };
   107→        const before = c.status;
   108→        const goalsBefore = c.hs + c.as;
   109→        advanceEvent(c, 1);
   110→        if (c.status === "EN_VIVO" && tick.current % 2 === 0) refreshOdds(c);
   111→        if (before !== "FINALIZADO" && c.status === "FINALIZADO" && c.kind !== "FUTBOL") settleWinnerEvent(c);
   112→        if (c.status === "EN_VIVO" && c.hs + c.as > goalsBefore) setFlash(c.id);
   113→        return c;
   114→      });
   115→      eventsRef.current = next;
   116→      setEvents(next);
   117→      if (tick.current % 3 === 0) saveEvents(next);
   118→      settleOpenTickets(next);
   119→      if (flash) setTimeout(() => setFlash(null), 2500);
   120→    }, 1000);
   121→    return () => clearInterval(iv);
   122→    // eslint-disable-next-line react-hooks/exhaustive-deps
   123→  }, []);
   124→
   125→  // ====== LIQUIDACION AUTOMATICA (pura: efectos fuera del updater) ======
   126→  const settleOpenTickets = useCallback((evs: BkEvent[]) => {
   127→    const map: Record<string, BkEvent> = {};
   128→    for (const e of evs) map[e.id] = e;
   129→    const prev = ticketsRef.current;
   130→    let changed = false;
   131→    let payoutDelta = 0, winsDelta = 0;
   132→    const next = prev.map((t) => {
   133→      if (t.status !== "ABIERTA") return t;
   134→      if (!t.legs.every((l) => map[l.eventId] && map[l.eventId].status === "FINALIZADO")) return t;
   135→      changed = true;
   136→      const won = t.legs.every((l) => choiceWon(map[l.eventId], l) === true);
   137→      if (won) {
   138→        // v84 FORTUNA: BONUS DE COMBINADA — más piernas, más premio (x3 +6% · x5 +12%)
   139→        const bonus = t.legs.length >= 5 ? 0.12 : t.legs.length >= 3 ? 0.06 : 0;
   140→        const pago = Math.round(t.potential * (1 + bonus));
   141→        payoutDelta += pago; winsDelta += 1;
   142→        addCoins(pago, `Apuesta GANADA BETNACION x${t.legs.length}${bonus > 0 ? ` · BONUS +${Math.round(bonus * 100)}%` : ""}`);
   143→        addXp(35);
   144→        recordBet(t.stake, true, pago);
   145→        sfx.coin();
   146→        toast.success(`¡APUESTA GANADA! +${pago} mon`, { description: `${t.legs.map((l) => `${l.choiceLabel} @${l.odd}`).join(" · ")}${bonus > 0 ? ` · BONUS COMBINADA +${Math.round(bonus * 100)}%` : ""}` });
   147→        // v84 FORTUNA: JACKPOT — combinada de 3+ con cuota total ≥ 8.0 tiene un
   148→        // 6% de llevarse el bote progresivo (se alimenta con el 4% de cada apuesta)
   149→        const cuotaTotal = t.stake > 0 ? t.potential / t.stake : 0;
   150→        if (t.legs.length >= 3 && cuotaTotal >= 8 && Math.random() < 0.06) {
   151→          const bote = jackpotRef.current;
   152→          addCoins(bote, "★ JACKPOT FORTUNA BETNACION ★");
   153→          saveJackpot(BK_JACKPOT_BASE);
   154→          setJackpot(BK_JACKPOT_BASE);
   155→          sfx.achievement();

   157→        }
   158→        return { ...t, status: "GANADA" as const, payout: pago, settledAt: Date.now() };
   159→      }
   160→      recordBet(t.stake, false, 0);
   161→      sfx.error();
   162→      toast.error("Apuesta perdida", { description: t.legs.map((l) => l.choiceLabel).join(" · ") });
   163→      return { ...t, status: "PERDIDA" as const, payout: 0, settledAt: Date.now() };
   164→    });

   166→    ticketsRef.current = next;
   167→    saveTickets(next);
   168→    setTickets(next);
   169→    if (payoutDelta > 0 || winsDelta > 0) {
   170→      setStats((s) => { const ns = { ...s, payout: s.payout + payoutDelta, wins: s.wins + winsDelta }; localStorage.setItem("vanguard-bk-stats", JSON.stringify(ns)); return ns; });

   172→  }, [addCoins, addXp, recordBet]);

   175→  const addLeg = (e: BkEvent, mKey: string, mLabel: string, cKey: string, cLabel: string, odd: number) => {
   176→    sfx.beep();

   180→      return [...filtered, leg].slice(0, 8);
   181→    });
   182→  };
   183→  const totalOdd = slip.reduce((a, l) => a * l.odd, 1);
   184→  const potential = Math.round(stake * totalOdd);
   185→
   186→  const placeBet = () => {
   187→    if (!slip.length) return;
   188→    if (stake < 10) { toast.error("Apuesta minima: 10 mon"); return; }
   189→    // v84 FORTUNA: APUESTA GRATIS DIARIA — la primera apuesta del día de hasta
   190→    // 75ⓒ la paga la casa: cero riesgo para el comandante

   192→    if (gratis) {
   193→      const nf = { date: hoyStr, used: true };
   194→      saveFreebet(nf);
   195→      setFreebet(nf);
   196→    } else {
   197→      if (coins < stake) { toast.error("Monedas insuficientes"); return; }
   198→      spendCoins(stake, `Apuesta BETNACION x${slip.length}`);
   199→    }
   200→    const t: BkTicket = {
   201→      id: `T-${Date.now()}-${Math.floor(Math.random() * 1e4)}`,
   202→      placedAt: Date.now(), legs: [...slip], stake,
   203→      potential: Math.round(stake * totalOdd), status: "ABIERTA", payout: 0,
   204→    };
   205→    const next = [t, ...loadTickets()];
   206→    saveTickets(next); setTickets(next); setSlip([]);
   207→    setStats((s) => { const ns = { ...s, staked: s.staked + stake, bets: s.bets + 1 }; localStorage.setItem("vanguard-bk-stats", JSON.stringify(ns)); return ns; });
   208→    // v84: el 4% de cada apuesta engorda el JACKPOT progresivo
   209→    const nj = jackpotRef.current + Math.round(stake * 0.04);
   210→    saveJackpot(nj);
   211→    setJackpot(nj);
   212→    sfx.coin();
   213→    toast.success(`Apuesta colocada · cuota ${totalOdd.toFixed(2)}${gratis ? " · ¡GRATIS!" : ""}`, { description: `Pago potencial: ${t.potential} mon` });

   223→    saveTickets(next); setTickets(next);
   224→    addCoins(val, "CASHOUT BETNACION");
   225→    recordBet(t.stake, val > t.stake, val);
   226→    sfx.coin();
   227→    toast.info(`CASHOUT cobrado: +${val} mon`, { description: val > t.stake ? "Cerraste con ganancia" : "Cerraste limitando perdida" });
   228→  };
