   357→            ))}
   358→            {/* cabeza + mandíbula */}
   359→            <circle cx="60" cy="38" r="17" fill={look.skin} />
   360→            {/* casco o boina */}
   361→            {look.headgear === "casco" && (
   362→              <>

   372→          </svg>

   374→        </div>
   375→

   380→              <Swatch key={c} color={c} activo={look.skin === c} onClick={() => setLook({ skin: c })} aria={`Piel ${c}`} />

   482→                  {r.id === "none" ? "—" : r.nombre.split(" ")[0]}

   484→              );
   485→            })}
   486→          </Fila>
   487→        </div>
   488→      </div>
   489→
   490→      {/* VITRINA DE MEDALLAS — se ganan, no se compran */}
   491→      <div className="mt-4 pt-3 border-t border-border/60">
   492→        <div className="flex items-center justify-between mb-2">
   493→          <div>
   494→            <h4 className="font-display font-black uppercase tracking-wide text-sm flex items-center gap-1.5">

   496→            </h4>

   500→          </div>
   501→        </div>

   503→          {MEDALLAS.map((m) => {
   504→            const got = ganadas.some((g) => g.id === m.id);
   505→            return (
   506→              <motion.div
   507→                key={m.id}
   508→                whileHover={got ? { scale: 1.04, rotate: -0.5 } : {}}

   513→              >

   515→                <svg viewBox="0 0 24 34" className="w-6 h-8 flex-shrink-0" aria-hidden>
   516→                  <rect x="4" y="2" width="16" height="14" fill={m.cinta[0]} />
   517→                  {m.cinta[1] && <rect x="4" y="5" width="16" height="4" fill={m.cinta[1]} />}
   518→                  {m.cinta[2] && <rect x="4" y="9" width="16" height="4" fill={m.cinta[2]} />}

   520→                    <>
   521→                      <circle cx="12" cy="24" r="7.5" fill={m.cinta[m.cinta.length - 1]} />
   522→                      <circle cx="12" cy="24" r="5" fill="none" stroke="#00000055" strokeWidth="1" />

   525→                  ) : (
   526→                    <circle cx="12" cy="24" r="7.5" fill="none" stroke="#555" strokeWidth="1.5" strokeDasharray="2 2" />
   527→                  )}
   528→                </svg>
   529→                <div className="min-w-0">
   530→                  <div className={cn("font-mono text-[9px] font-black uppercase leading-tight", got ? "text-amber" : "text-muted-foreground")}>{m.nombre}</div>
   531→                  <div className="font-mono text-[8px] text-muted-foreground leading-tight">{m.descripcion}</div>
   532→                </div>
   533→              </motion.div>
   534→            );
   535→          })}
   536→        </div>
   537→      </div>
   538→    </div>
   539→  );
   540→}

   573→function Compare({ label, mine, global, fmt }: { label: string; mine: number; global: number; fmt: (v: number) => string }) {
   574→  const max = Math.max(mine, global, 1);
   575→  const better = mine >= global;
   576→  return (
   577→    <div className="mb-2">
   578→      <div className="flex justify-between font-mono text-[9px] mb-0.5">
   579→        <span>{label}</span>
   580→        <span className={better ? "text-neon" : "text-crisis"}>{fmt(mine)} vs {fmt(global)}</span>
   581→      </div>
   582→      <div className="h-1.5 bg-secondary relative">
   583→        <div className={cn("absolute left-0 top-0 h-full", better ? "bg-neon" : "bg-crisis")} style={{ width: `${(mine / max) * 100}%` }} />
   584→        <div className="absolute top-[-2px] bottom-[-2px] w-0.5 bg-white/70" style={{ left: `${(global / max) * 100}%` }} title="promedio global" />
   585→      </div>
   586→    </div>
   587→  );
   588→}
