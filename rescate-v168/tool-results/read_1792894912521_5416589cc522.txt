    71→function uidLocal(): string {
    72→  try {
    73→    let id = localStorage.getItem("vanguard-mp-uid");
    74→    if (!id) {
    75→      id = `mp-${crypto.randomUUID?.() ?? Math.random().toString(36).slice(2)}`;
    76→      localStorage.setItem("vanguard-mp-uid", id);
    77→    }
    78→    return id;
    79→  } catch {
    80→    return `anon-${Math.random().toString(36).slice(2)}`;
    81→  }
    82→}
    83→
    84→function generarGabinete(): Ministro[] {
    85→  const pick = (arr: string[]) => arr[Math.floor(Math.random() * arr.length)];
    86→  return ROLES.map((rol) => ({ rol, nombre: `${pick(NOMBRES)} ${pick(APELLIDOS)}` }));
    87→}

    96→          <stop offset="0%" stopColor="#ffffff" stopOpacity="0.34" />
    97→          <stop offset="38%" stopColor="#ffffff" stopOpacity="0.06" />

   100→        </linearGradient>

   128→// ---------- panel ----------
   129→export function MiPaisPanel() {
   130→  const [uid, setUid] = useState("");
   131→  const [mia, setMia] = useState<Nacion | null>(null);
   132→  const [top, setTop] = useState<Nacion[]>([]);
   133→  const [uniones, setUniones] = useState<UnionRow[]>([]);
   134→  const [invitaciones, setInvitaciones] = useState<Invitacion[]>([]);
   135→  const [cargando, setCargando] = useState(true);
   136→
   137→  // wizard
   138→  const [paso, setPaso] = useState(1);
   139→  const [f, setF] = useState({
   140→    nombre: "", gentilicio: "", lema: "",
   141→    c1: "#c8102e", c2: "#ffd60a", simbolo: "★",
   142→    gobierno: "REPÚBLICA", region: "América del Sur", capital: "",
   143→    presidente: "", ministros: [] as Ministro[],
   144→  });
   145→  const [guardando, setGuardando] = useState(false);
   146→
   147→  // unión
   148→  const [uNombre, setUNombre] = useState("");
   149→  const [uTipo, setUTipo] = useState("ALIANZA");
   150→  const [uLema, setULema] = useState("");
   151→  const [reclutando, setReclutando] = useState(false);
   152→  const vivoRef = useRef(true);
   153→
   154→  const refrescar = useCallback(async (u: string) => {
   155→    try {
   156→      const r = await fetch(`/api/naciones?uid=${encodeURIComponent(u)}`, { cache: "no-store" });
   157→      const d = await r.json();
   158→      if (!vivoRef.current) return;
   159→      setMia(d?.mia ?? null);
   160→      setTop(Array.isArray(d?.top) ? d.top : []);
   161→      setUniones(Array.isArray(d?.uniones) ? d.uniones : []);
   162→    } catch { /* aguanta */ }
   163→  }, []);
   164→
   165→  useEffect(() => {
   166→    vivoRef.current = true;
   167→    const u = uidLocal();
   168→    setUid(u);
   169→    refrescar(u);
   170→    setCargando(false);
   171→    return () => { vivoRef.current = false; };
   172→  }, [refrescar]);
   173→
   174→  // buzón de invitaciones (polling cada 12s)
   175→  useEffect(() => {
   176→    if (!uid) return;
   177→    let alive = true;
   178→    const poll = async () => {
   179→      try {
   180→        const r = await fetch(`/api/reclutar?uid=${encodeURIComponent(uid)}`, { cache: "no-store" });
   181→        const d = await r.json();
   182→        if (alive) setInvitaciones(Array.isArray(d?.items) ? d.items : []);
   183→      } catch { /* silencio */ }
   184→    };
   185→    poll();
   186→    const t = setInterval(poll, 12_000);
   187→    return () => { alive = false; clearInterval(t); };
   188→  }, [uid]);
   189→
   190→  const proclamar = async () => {
   191→    if (f.nombre.trim().length < 2) { toast.error("Tu país necesita un nombre"); setPaso(1); return; }
   192→    setGuardando(true);
   193→    try {
   194→      const r = await fetch("/api/naciones", {
   195→        method: "POST",
   196→        headers: { "Content-Type": "application/json" },
   197→        body: JSON.stringify({ action: "guardar", uid, nacion: f }),
   198→      });
   199→      const d = await r.json();
   200→      if (d?.ok) {
   201→        setMia(d.mia);
   202→        toast.success(`¡${f.nombre} proclamado! Ya figura en el mapa de naciones.`);
   203→        refrescar(uid);
   204→      } else {
   205→        toast.error(d?.error || "No se pudo proclamar la nación");
   206→      }
   207→    } catch {
   208→      toast.error("Red intermitente — intenta de nuevo");
   209→    } finally {
   210→      setGuardando(false);
   211→    }
   212→  };
   213→
   214→  const accionUnion = async (action: string, union?: string) => {
   215→    try {
   216→      const r = await fetch("/api/naciones", {
   217→        method: "POST",
   218→        headers: { "Content-Type": "application/json" },
   219→        body: JSON.stringify({ action, uid, union: union ?? { nombre: uNombre, tipo: uTipo, lema: uLema } }),
   220→      });
   221→      const d = await r.json();
   222→      if (d?.ok) {
   223→        toast.success(action === "crear_union" ? `¡${d.union} fundada!` : action === "salir" ? "Saliste de la unión" : `Te uniste a ${d.union}`);
   224→        setUNombre(""); setULema("");
   225→        refrescar(uid);
   226→      } else {
   227→        toast.error(d?.error || "No se pudo completar");
   228→      }
   229→    } catch {
   230→      toast.error("Red intermitente — intenta de nuevo");
   231→    }
   232→  };
   233→
   234→  const reclutar = async () => {
   235→    setReclutando(true);
   236→    try {
   237→      const r = await fetch("/api/reclutar", {
   238→        method: "POST",
   239→        headers: { "Content-Type": "application/json" },
   240→        body: JSON.stringify({ action: "invitar", uid }),
   241→      });
   242→      const d = await r.json();
   243→      if (d?.ok) toast.success(`Invitación enviada a un guerrero en línea (${String(d.objetivo).slice(0, 10)}…). Si acepta, se une a tu causa.`);
   244→      else toast.error(d?.error || "No se pudo enviar la invitación");
   245→    } catch {
   246→      toast.error("Red intermitente — intenta de nuevo");
   247→    } finally {
   248→      setReclutando(false);
   249→    }
   250→  };
   251→
   252→  const aceptar = async (inv: Invitacion) => {
   253→    try {
   254→      const r = await fetch("/api/reclutar", {
   255→        method: "POST",
   256→        headers: { "Content-Type": "application/json" },
   257→        body: JSON.stringify({ action: "aceptar", uid, id: inv.id }),
   258→      });
   259→      const d = await r.json();
   260→      setInvitaciones((prev) => prev.filter((i) => i.id !== inv.id));
   261→      if (d?.ok) {
   262→        toast.success(inv.union ? `¡Bienvenido a ${inv.union}!` : `Te aliaste con ${inv.nacion}`);
   263→        refrescar(uid);
   264→      }
   265→    } catch { /* silencio */ }
   266→  };
   267→
   268→  // ---------- render ----------
   269→  return (
   270→    <div className="space-y-4">
   271→      <PanelHeader title="MI PAÍS" subtitle="Simulador de nación" />
   272→
   273→      <TituloEpico
   274→        titulo="MI PAÍS"
   275→        volanta="Simulador de nación — funda, une y conquista"
   276→        imagen="/ilustraciones/mi-pais.jpg"
   277→        escena="ciudad"
   278→        semilla={88}
   279→        texto="Proclama tu propia nación: ponle nombre y bandera, elige su gobierno y territorio, nombra a sus figuras públicas, fúndala en una unión o cartel… y recluta aliados al azar entre los guerreros que están en línea ahora mismo."
   280→        altura={270}
   281→      />
   282→
   283→      {cargando ? (
   284→        <div className="hud-panel p-8 flex items-center justify-center gap-2 text-muted-foreground">
   285→          <Loader2 className="w-4 h-4 animate-spin" /> Cargando tu expediente de fundador…
   286→        </div>
   287→      ) : !mia ? (
   288→        /* ==================== WIZARD DE CREACIÓN ==================== */
   289→        <section className="hud-panel p-4 md:p-5" aria-label="Crear mi país">
   290→          {/* pasos */}
   291→          <div className="flex items-center gap-1.5 mb-4" aria-hidden>
   292→            {[1, 2, 3, 4].map((n) => (
   293→              <div key={n} className={cn("h-1 flex-1 rounded-full transition-all duration-500", paso >= n ? "bg-gradient-to-r from-[#ffc46b] to-[#ff8a2a] shadow-[0_0_8px_rgba(255,138,42,0.5)]" : "bg-white/10")} />
   294→            ))}
   295→          </div>
   296→          <p className="font-mono text-[10px] uppercase tracking-[0.3em] text-amber mb-3">PASO {paso} DE 4</p>
   297→
   298→          {paso === 1 ? (
   299→            <div className="space-y-3.5">
   300→              <h3 className="font-display font-bold uppercase text-sm flex items-center gap-2"><Flag className="w-4 h-4 text-amber" /> Identidad de la nación</h3>
   301→              <input value={f.nombre} onChange={(e) => setF({ ...f, nombre: e.target.value.slice(0, 32) })} placeholder="Nombre del país (ej. NUEVA ESPERANZA)" aria-label="Nombre del país" className="w-full bg-black/40 border border-white/10 focus:border-amber-hud/60 rounded-sm px-3 py-2.5 text-sm outline-none transition-colors min-h-[44px]" />
   302→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   303→                <input value={f.gentilicio} onChange={(e) => setF({ ...f, gentilicio: e.target.value.slice(0, 24) })} placeholder="Gentilicio (ej. esperanzano)" aria-label="Gentilicio" className="w-full bg-black/40 border border-white/10 focus:border-amber-hud/60 rounded-sm px-3 py-2.5 text-sm outline-none transition-colors min-h-[44px]" />
   304→                <input value={f.capital} onChange={(e) => setF({ ...f, capital: e.target.value.slice(0, 28) })} placeholder="Capital (ej. Puerto Libre)" aria-label="Capital" className="w-full bg-black/40 border border-white/10 focus:border-amber-hud/60 rounded-sm px-3 py-2.5 text-sm outline-none transition-colors min-h-[44px]" />
   305→              </div>
   306→              <input value={f.lema} onChange={(e) => setF({ ...f, lema: e.target.value.slice(0, 80) })} placeholder="Lema nacional (ej.    307→            </div>
   308→          ) : null}
   309→
   310→          {paso === 2 ? (
   311→            <div className="space-y-3.5">
   312→              <h3 className="font-display font-bold uppercase text-sm flex items-center gap-2"><Flag className="w-4 h-4 text-amber" /> Bandera nacional</h3>
   313→              <div className="flex items-center gap-4 flex-wrap">
   314→                <Bandera c1={f.c1} c2={f.c2} simbolo={f.simbolo} className="w-40 h-auto rounded-sm shadow-[0_10px_28px_rgba(0,0,0,0.6)]" />
   315→                <div className="flex items-center gap-3">
   316→                  <label className="flex flex-col gap-1 font-mono text-[10px] uppercase tracking-widest text-muted-foreground">
   317→                    Tinta 1
   318→                    <input type="color" value={f.c1} onChange={(e) => setF({ ...f, c1: e.target.value })} aria-label="Color 1 de la bandera" className="w-12 h-10 bg-transparent cursor-pointer" />
   319→                  </label>
   320→                  <label className="flex flex-col gap-1 font-mono text-[10px] uppercase tracking-widest text-muted-foreground">
   321→                    Tinta 2
   322→                    <input type="color" value={f.c2} onChange={(e) => setF({ ...f, c2: e.target.value })} aria-label="Color 2 de la bandera" className="w-12 h-10 bg-transparent cursor-pointer" />

   324→                </div>
   325→              </div>

   331→                      f.simbolo === s ? "border-amber-hud bg-amber-hud/20 shadow-[0_0_12px_rgba(255,138,42,0.3)]" : "border-white/10 bg-black/30 hover:border-amber-hud/50")}>
   332→                    {s}
   333→                  </button>
   334→                ))}
   335→              </div>
   336→            </div>
   337→          ) : null}
   338→
   339→          {paso === 3 ? (
   340→            <div className="space-y-3.5">
   341→              <h3 className="font-display font-bold uppercase text-sm flex items-center gap-2"><Landmark className="w-4 h-4 text-amber" /> Gobierno y territorio</h3>

   344→                {GOBIERNOS.map((g) => (
   345→                  <button key={g} onClick={() => setF({ ...f, gobierno: g })} aria-pressed={f.gobierno === g}

"Nombre del Rey / la Reina" : f.gobierno === "CARTEL" ? "Nombre del Jefe" : "Nombre del Presidente"} aria-label="Nombre del líder" className="w-full bg-black/40 border border-white/10 focus:border-amber-hud/60 rounded-sm px-3 py-2.5 text-sm outline-none transition-colors min-h-[44px]" />
