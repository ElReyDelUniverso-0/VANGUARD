    88→function altDePunto(c: Contacto): number {
    89→  switch (c.tipo) {
    90→    case "satelite": return 0.24 + Math.min(0.3, (c.altKm - 420) / 1400);
    91→    case "avion": return 0.075;
    92→    case "buque": return 0.015;
    93→    case "dron": return 0.045;
    94→    default: return 0.02;
    95→  }
    96→}
