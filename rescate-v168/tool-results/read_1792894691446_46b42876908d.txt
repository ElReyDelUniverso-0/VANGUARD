   103→function useMyMpId() {
   104→  return useMemo(() => {
   105→    if (typeof window === "undefined") return "srv";
   106→    let id = localStorage.getItem("vanguard-mp-uid");
   107→    if (!id) {
   108→      id = `mp-${crypto.randomUUID?.() ?? Math.random().toString(36).slice(2)}`;
   109→      localStorage.setItem("vanguard-mp-uid", id);
   110→    }

   119→function useMpConnected() {
   120→  // v21: valor inicial perezoso (sin setState síncrono en el effect)
   121→  const [connected, setConnected] = useState(() => !!peekRealtime()?.connected);
   122→  useEffect(() => {
   123→    const s = getRealtime();
   124→    if (!s) return;
   125→    const on = () => setConnected(true);
   126→    const off = () => setConnected(false);
   127→    s.on("connect", on);
   128→    s.on("disconnect", off);
   129→    return () => {
   130→      s.off("connect", on);
   131→      s.off("disconnect", off);
   132→    };
   133→  }, []);
   134→  return { connected, retry: () => getRealtime()?.connect() };
   135→}

   223→// MODO GUERRA (v7 + chat v17 + revancha v17 + partida rápida v17)
   224→// ============================================================
   225→function GuerraMode({ alias }: { alias: string }) {
   226→  const recordMpCapture = useGameStore((s) => s.recordMpCapture);
   227→  const recordMpWin = useGameStore((s) => s.recordMpWin);
   228→  const mpStats = useGameStore((s) => s.mpStats);
   229→  const myId = useMyMpId();
   230→  const { t } = useT();
   231→  // v23 PERSONAJES-PAÍS: la countryball del perfil representa al comandante en la guerra
   232→  const cbAvatar = useProfileStore((s) => s.cbAvatar);
   233→  const setCbAvatar = useProfileStore((s) => s.setCbAvatar);
   234→
   235→  const [state, setState] = useState<MpState | null>(null);
   236→  const [connected, setConnected] = useState(false);
   237→  const [selected, setSelected] = useState<string | null>(null);
   238→  const [now, setNow] = useState(Date.now());
   239→  const [chatDraft, setChatDraft] = useState("");
   240→  const [myElo, setMyElo] = useState<number | null>(null);
   241→  const lastBattleSeq = useRef(0);
   242→  const winnerSeqSeen = useRef(0);
   243→  const participatedRef = useRef(false);
   244→  const chatInputRef = useRef<HTMLInputElement | null>(null);
   245→  const chatListRef = useRef<HTMLDivElement | null>(null);
   246→  const myNameRef = useRef(alias || "OPERADOR");
   247→  const myAvatarRef = useRef(cbAvatar); // v23: avatar país en el handshake
   248→  const dropRef = useRef(false); // v21: avisa al reconectar si hubo caída
   249→  const [slowNet, setSlowNet] = useState(false); // v21: aviso de servidor lento/caído
   250→  useEffect(() => { myNameRef.current = alias || "OPERADOR"; }, [alias]);
   251→  useEffect(() => { myAvatarRef.current = cbAvatar; }, [cbAvatar]);
   252→  useEffect(() => {
   253→    const t = setTimeout(() => setSlowNet(true), 8000);
   254→    return () => clearTimeout(t);
   255→  }, []);
   256→
   257→  // ===== reloj local para cuentas atras =====
   258→  useEffect(() => {
   259→    const t = setInterval(() => setNow(Date.now()), 1000);
   260→    return () => clearInterval(t);
   261→  }, []);
   262→
   263→  // ===== ELO propio para la cabecera =====
   264→  useEffect(() => {
   265→    const name = (alias || "").toUpperCase().slice(0, 18);
   266→    if (!name) return;
   267→    fetch(`/api/mp/stats?alias=${encodeURIComponent(name)}`)
   268→      .then((r) => (r.ok ? r.json() : null))
   269→      .then((d) => { if (d?.profile) setMyElo(d.profile.elo); })
   270→      .catch(() => {});
   271→  }, [alias]);
   272→
   273→  // ===== conexion y estado del juego =====
   274→  useEffect(() => {
   275→    const socket = getRealtime();
   276→    if (!socket) return;
   277→
   278→    const onConnect = () => {
   279→      setConnected(true);
   280→      if (dropRef.current) {
   281→        dropRef.current = false;
   282→        setSlowNet(false);
   283→        toast.success("Reconectado a la guerra global — sincronizando estado…");
   284→      }
   285→      socket.emit("mp:join", { playerId: myId, name: myNameRef.current, avatar: myAvatarRef.current }, (res: { state?: MpState } | undefined) => {
   286→        if (res?.state) setState(res.state);
   287→      });
   288→    };
   289→    const onDisconnect = () => {
   290→      setConnected(false);
   291→      dropRef.current = true;
   292→      toast.warning("Conexión perdida — reconexión automática en marcha…");
   293→    };
   294→    const onState = (s: MpState) => {
   295→      setState(s);
   296→      // recompensas por captura propia (una sola vez por batalla)
   297→      const lb = s.lastBattle;
   298→      if (lb && lb.seq > lastBattleSeq.current) {
   299→        lastBattleSeq.current = lb.seq;
   300→        if (lb.attacker === myId) {
   301→          participatedRef.current = true;
   302→          if (lb.captured) {
   303→            recordMpCapture(2);
   304→            postMpResult(myNameRef.current, "CAPTURE");
   305→            toast.success(`¡Territorio capturado! +2 gemas, +5 monedas`);
   306→            sfx.unlock();
   307→          } else {
   308→            toast.error(`Asalto repelido en defensa enemiga`);
   309→            sfx.error();
   310→          }
   311→        }
   312→      }
   313→      // victoria propia + registro de resultado en el ranking ELO
   314→      if (s.winnerSeq > winnerSeqSeen.current) {
   315→        winnerSeqSeen.current = s.winnerSeq;
   316→        if (participatedRef.current) {
   317→          const iWon = s.winnerId === myId;
   318→          postMpResult(myNameRef.current, iWon ? "MP_WIN" : "MP_LOSS");
   319→          if (iWon) {
   320→            recordMpWin();
   321→            toast.success("DOMINIO GLOBAL conseguido: +25 gemas, +200 monedas · ELO +30");
   322→          } else {
   323→            toast.info(`Victoria de ${s.winnerName ?? "?"} — resultado registrado en el ranking`);
   324→          }
   325→        } else if (s.winnerId === myId) {
   326→          recordMpWin();
   327→          toast.success("DOMINIO GLOBAL conseguido: +25 gemas, +200 monedas");
   328→        }
   329→      }
   330→      // autoscroll del chat
   331→      requestAnimationFrame(() => {
   332→        const el = chatListRef.current;
   333→        if (el) el.scrollTop = el.scrollHeight;
   334→      });
   335→    };
   336→
   337→    socket.on("connect", onConnect);
   338→    socket.on("disconnect", onDisconnect);
   339→    socket.on("mp:state", onState);
   340→    if (socket.connected) onConnect();
   341→
   342→    return () => {
   343→      socket.off("connect", onConnect);
   344→      socket.off("disconnect", onDisconnect);
   345→      socket.off("mp:state", onState);
   346→    };
   347→  }, [myId, recordMpCapture, recordMpWin]);
   348→
   349→  const me = state?.players[myId];
   350→  const myTerritories = useMemo(
   351→    () => (state ? state.territoryMeta.filter((t) => state.territories[t.id]?.owner === myId) : []),
   352→    [state, myId]
   353→  );
   354→  // v23: nombre -> avatar país para el chat y banners (hook antes del early return)
   355→  const avatarByName = useMemo(() => {
   356→    const m = new Map<string, string>();
   357→    if (state) for (const p of Object.values(state.players)) if (p.avatar) m.set(p.name, p.avatar);
   358→    return m;
   359→  }, [state]);
   360→  const claimedByMe = myTerritories.length > 0;
   361→  const claimable = state?.phase === "LOBBY";
   362→
   363→  // objetivos de ataque: adyacentes al seleccionado que no son mios
   364→  const attackTargets = useMemo(() => {
   365→    if (!state || !selected) return [];
   366→    const t = state.territoryMeta.find((x) => x.id === selected);
   367→    if (!t) return [];
   368→    return t.adj.filter((a) => state.territories[a] && state.territories[a].owner !== myId);
   369→  }, [state, selected, myId]);
   370→
   371→  const emit = (event: string, payload: unknown, cb?: (r: { ok: boolean; reason?: string }) => void) => {
   372→    const socket = getRealtime();
   373→    socket.emit(event, payload, cb);
   374→  };
   375→
   376→  const claim = (terrId: string) => {
   377→    emit("mp:claim", { terrId }, (r) => {
   378→      if (r?.ok) { participatedRef.current = true; toast.success(`Base establecida. La partida arranca en segundos`); sfx.unlock(); }
   379→      else toast.error(r?.reason ?? "No se pudo reclamar");
   380→    });
   381→  };
   382→
   383→  // v17: PARTIDA RAPIDA — reclama un territorio libre al azar
   384→  const quickMatch = () => {
   385→    if (!state) return;
   386→    const free = state.territoryMeta.filter((t) => !state.territories[t.id]?.owner);
   387→    if (free.length === 0) { toast.error("No quedan territorios libres"); return; }
   388→    claim(free[Math.floor(Math.random() * free.length)].id);
   389→  };
   390→
   391→  const deploy = (terrId: string, qty = 1) => {
   392→    emit("mp:deploy", { terrId, qty }, (r) => {
   393→      if (!r?.ok) toast.error(r?.reason ?? "Despliegue rechazado");
   394→      else { participatedRef.current = true; sfx.click(); }
   395→    });
   396→  };
   397→
   398→  const deployAll = () => {
   399→    emit("mp:deployAll", {}, (r) => {
   400→      if (r?.ok) { participatedRef.current = true; toast.success("Reservas desplegadas en tus territorios"); sfx.success(); }
   401→      else toast.error(r?.reason ?? "Nada que desplegar");
   402→    });
   403→  };
   404→
   405→  const attack = (from: string, to: string) => {
   406→    emit("mp:attack", { from, to }, (r) => {
   407→      if (!r?.ok) toast.error(r?.reason ?? "Ataque invalido");
   408→      else participatedRef.current = true;
   409→    });
   410→    setSelected(null);
   411→  };
   412→
   413→  const sendChat = () => {
   414→    const body = chatDraft.trim();
   415→    if (!body) return;
   416→    emit("mp:chat", { body }, (r) => {
   417→      if (!r?.ok) toast.error(r?.reason ?? "No se pudo enviar");
   418→      else setChatDraft("");
   419→    });
   420→  };
   421→
   422→  const requestRematch = () => {
   423→    emit("mp:rematch", {}, (r) => {
   424→      if (!r?.ok) toast.error(r?.reason ?? "No disponible");
   425→      else sfx.click();
   426→    });
   427→  };
   428→
   429→  // v23: elige tu personaje-país y sincronízalo con la partida en vivo
   430→  const pickCharacter = (code: string) => {
   431→    setCbAvatar(code);
   432→    myAvatarRef.current = code;
   433→    emit("mp:join", { playerId: myId, name: myNameRef.current, avatar: code }, () => {});
   434→    sfx.unlock();
   435→    toast.success(`${t("mp.characterSet")}: ${countryName(code)}`);
   436→  };
   437→
   438→  const secondsLeft = (target: number) => Math.max(0, Math.ceil((target - now) / 1000));
   439→
   440→  // ================== RENDER ===================
   441→  if (!state) {
   442→    return (
   443→      <div className="hud-corner p-10 text-center">
   444→        {connected ? (
   445→          <>
   446→            <RefreshCw className="w-8 h-8 mx-auto mb-3 text-amber animate-spin" />
   447→            <p className="text-sm font-mono text-muted-foreground uppercase">
   448→              Sincronizando partida global...
   449→            </p>
   450→          </>
   451→        ) : (
   452→          <>
   453→            <WifiOff className="w-8 h-8 mx-auto mb-3 text-red-hud" />
   454→            <p className="text-sm font-mono text-muted-foreground uppercase">
   455→              {slowNet ? "Servidor de partidas en mantenimiento" : "Conectando al servidor de partidas :3003..."}
   456→            </p>
   457→            <p className="text-[10px] font-mono text-muted-foreground/70 uppercase mt-1 mb-4">
   458→              reintentos automáticos cada 5s · tu progreso está a salvo
   459→            </p>
   460→            <Button
   461→              size="sm"
   462→              variant="outline"
   463→              onClick={() => getRealtime()?.connect()}
   464→              className="font-mono text-[10px] uppercase tracking-widest border-cyan-hud text-cyan-hud hover:bg-cyan-hud/10"
   465→            >
   466→              Reintentar ahora
   467→            </Button>
   468→          </>
   469→        )}
   470→      </div>
   471→    );

   839→
   840→function DueloMode({ alias }: { alias: string }) {
   841→  const addCoins = useGameStore((s) => s.addCoins);
   842→  const addGems = useGameStore((s) => s.addGems);
   843→  const addXp = useGameStore((s) => s.addXp);
   844→  const myId = useMyMpId();
   845→
   846→  const [duel, setDuel] = useState<DuelState | null>(null);
   847→  const [searching, setSearching] = useState(false);
   848→  const [chosen, setChosen] = useState<{ q: number; idx: number } | null>(null);
   849→  const rewardedRef = useRef<number>(0);
   850→  const myNameRef = useRef(alias || "OPERADOR");
   851→  useEffect(() => { myNameRef.current = alias || "OPERADOR"; }, [alias]);
   852→
   853→  // suscripcion al duelo
   854→  useEffect(() => {
   855→    const socket = getRealtime();
   856→    if (!socket) return;
   857→    const onState = (s: DuelState) => {
   858→      setDuel(s);
   859→      setSearching(false);
   860→      if (s.phase === "QUESTION" && s.me && !s.me.answered) {
   861→        // nueva pregunta disponible
   862→      }
   863→      if (s.phase === "ENDED" && s.result && rewardedRef.current !== s.id) {
   864→        rewardedRef.current = s.id;
   865→        const { outcome, walkover } = s.result;
   866→        if (outcome === "WIN") {
   867→          addCoins(120, "Duelo 1v1 ganado");
   868→          addGems(2, "Duelo 1v1 ganado");
   869→          addXp(80);
   870→          toast.success(walkover ? "Rival desconectado — victoria por abandono: +120 mon, +2 gemas" : "DUELO GANADO: +120 monedas, +2 gemas, +80 XP · ELO +22");
   871→          sfx.achievement?.();
   872→        } else if (outcome === "LOSS") {
   873→          addCoins(30, "Duelo 1v1 participado");
   874→          addXp(20);
   875→          toast.info(walkover ? "Abandonaste el duelo — premio de consuelo: +30 mon" : "Duelo perdido — premio de participación: +30 mon, +20 XP");
   876→        } else {
   877→          addCoins(60, "Duelo 1v1 empatado");
   878→          addXp(40);
   879→          toast.info("EMPATE — +60 monedas, +40 XP · ELO +5");
   880→        }
   881→        postMpResult(myNameRef.current,
   882→          outcome === "WIN" ? "DUEL_WIN" : outcome === "LOSS" ? "DUEL_LOSS" : "DUEL_TIE",
   883→          { streak: outcome === "WIN" ? (s.me.correct ?? 0) : 0 });
   884→      }
   885→    };
   886→    const onClosed = () => { setDuel(null); setSearching(false); };
   887→    socket.on("duel:state", onState);

  1110→// barra de progreso del tiempo (10Hz, solo este componente se actualiza)
  1111→function DuelTimerBar({ phase, ends, duration }: { phase: DuelState["phase"]; ends: number; duration: number }) {
  1112→  const [now, setNow] = useState(Date.now());
  1113→  useEffect(() => {
  1114→    if (phase === "ENDED") return;
  1115→    const t = setInterval(() => setNow(Date.now()), 100);
  1116→    return () => clearInterval(t);
  1117→  }, [phase]);
  1118→  const frac = Math.max(0, Math.min(1, (ends - now) / duration));
  1119→  const urgent = frac < 0.3 && phase === "QUESTION";
  1120→  return (
  1121→    <div className="h-1.5 w-full bg-black/60 border border-border overflow-hidden rounded-sm" role="progressbar" aria-label="tiempo restante">
  1122→      <div
  1123→        className={cn("h-full transition-[width] duration-100 ease-linear", urgent ? "bg-red-hud" : "bg-gradient-to-r from-electric to-neon")}
  1124→        style={{ width: `${frac * 100}%` }}
  1125→      />
  1126→    </div>
  1127→  );
  1128→}
