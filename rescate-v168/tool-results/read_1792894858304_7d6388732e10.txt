    46→function loadJSON<T>(key: string, fallback: T): T {
    47→  try {
    48→    const raw = localStorage.getItem(key);
    49→    if (raw) return { ...fallback, ...(JSON.parse(raw) as object) } as T;
    50→  } catch { /* noop */ }
    51→  return fallback;
    52→}
    53→
    54→function saveJSON(key: string, v: unknown) {
    55→  try { localStorage.setItem(key, JSON.stringify(v)); } catch { /* noop */ }
    56→}

    58→// envía con el BOT PERSONAL del usuario (directo, sin servidor)
    59→async function enviarPersonal(bot: OwnBot, text: string): Promise<void> {
    60→  const res = await fetch(`https://api.telegram.org/bot${bot.token.trim()}/sendMessage`, {
    61→    method: "POST",
    62→    headers: { "Content-Type": "application/json" },
    63→    body: JSON.stringify({
    64→      chat_id: bot.chatId.trim(),
    65→      text: text.slice(0, 3900),
    66→      parse_mode: "HTML",
    67→      disable_web_page_preview: false,
    68→    }),
    69→    signal: AbortSignal.timeout(9000),
    70→  });
    71→  const data = (await res.json()) as { ok?: boolean; description?: string };
    72→  if (!data.ok) throw new Error(data.description ?? "Telegram rechazó el envío");
    73→}

    75→function huella(s: string): string {
    76→  return s.toLowerCase().replace(/[^a-z0-9áéíóúñü ]/g, "").split(/\s+/).slice(0, 9).join(" ");
    77→}
