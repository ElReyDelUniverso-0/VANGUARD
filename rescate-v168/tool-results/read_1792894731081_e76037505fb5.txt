    65→    65→    if (mostrandoPorQue) return;

    67→    67→    setMostrandoPorQue(true);
    68→    68→    if (idx === preg.best) sfx.success();

    73→    73→    const gain = elegida === preg.best ? 1 : 0;
    74→    74→    const next = { ...scores, [preg.apt]: scores[preg.apt] + gain };

    76→    76→    setMostrandoPorQue(false);

   201→   201→function TurnoActivo({ depto, tareas, onFin }: { depto: Departamento; tareas: TareaTurno[]; onFin: (aciertos: number) => void }) {
   202→   202→  const [paso, setPaso] = useState(0);
   203→   203→  const [aciertos, setAciertos] = useState(0);
   204→   204→  const [elegida, setElegida] = useState<number | null>(null);
   205→   205→  const c = COLOR_CLS[depto.color];
   206→   206→  const t = tareas[paso];
   207→   207→
   208→   208→  const responder = (i: number) => {
   209→   209→    if (elegida !== null) return;
   210→   210→    setElegida(i);
   211→   211→    if (i === t.best) {
   212→   212→      setAciertos((a) => a + 1);
   213→   213→      sfx.success();
   214→   214→    } else sfx.error();
   215→   215→    setTimeout(() => {
   216→   216→      if (paso + 1 >= tareas.length) onFin(aciertos + (i === t.best ? 1 : 0));
   217→   217→      else { setPaso((p) => p + 1); setElegida(null); }
   218→   218→    }, 750);
   219→   219→  };
   220→   220→
   221→   221→  return (
   222→   222→    <div className={cn("hud-corner p-3 border", c.border)}>
   223→   223→      <div className="flex items-center justify-between mb-2">
   224→   224→        <p className={cn("text-[10px] font-mono uppercase tracking-wider flex items-center gap-1.5", c.text)}>
   225→   225→          <Zap className="w-3.5 h-3.5" /> TURNO EN CURSO · {depto.nombre}
   226→   226→        </p>
   227→   227→        <span className="text-[10px] font-mono text-muted-foreground">tarea {paso + 1}/{tareas.length}</span>
   228→   228→      </div>
   229→   229→      <h4 className="text-sm font-bold text-foreground leading-snug mb-2.5">{t.q}</h4>
   230→   230→      <div className="space-y-1.5">
   231→   231→        {t.opts.map((opt, i) => (
   232→   232→          <button
   233→   233→            key={i}
   234→   234→            onClick={() => responder(i)}
   235→   235→            disabled={elegida !== null}
   236→   236→            className={cn(
   237→   237→              "w-full text-left border px-2.5 py-2 text-xs transition-colors",
   238→   238→              elegida === null && "border-border hover:border-amber-hud cursor-pointer",
   239→   239→              elegida !== null && i === t.best && "border-green-hud bg-green-hud/15",
   240→   240→              elegida !== null && i === elegida && i !== t.best && "border-red-hud bg-red-hud/10",
   241→   241→              elegida !== null && i !== t.best && i !== elegida && "border-border/40 text-muted-foreground/60"
   242→   242→            )}
   243→   243→          >
   244→   244→            {opt}
   245→   245→          </button>
   246→   246→        ))}
   247→   247→      </div>
   248→   248→    </div>
   249→   249→  );
   250→   250→}

   253→   253→export function EmpleosPanel() {
   254→   254→  const coins = useGameStore((s) => s.coins);
   255→   255→  const addCoins = useGameStore((s) => s.addCoins);
   256→   256→  const addXp = useGameStore((s) => s.addXp);
   257→   257→  const [st, setSt] = useState<EmpleosState>(() => loadEmpleos());
   258→   258→  const [tareaAbierta, setTareaAbierta] = useState(false);
   259→   259→  const [resumenTurno, setResumenTurno] = useState<{ aciertos: number; pago: { base: number; propina: number; bonoPerfecta: number; total: number; xpGanada: number; perfecta: boolean } } | null>(null);
   260→   260→  const [tick, setTick] = useState(0);
   261→   261→  const [confirmarRenuncia, setConfirmarRenuncia] = useState(false);
   262→   262→  const [oferta, setOferta] = useState<{ depto: Departamento; scores: Record<Aptitud, number> } | null>(null);
   263→   263→  const stRef = useRef(st);
   264→   264→  stRef.current = st;
   265→   265→
   266→   266→  const depto = useMemo(() => DEPARTAMENTOS.find((d) => d.id === st.depto) ?? null, [st.depto]);
   267→   267→
   268→   268→  // nómina corre en vivo: acumula al abrir, cobre con tictac visual cada 1 s
   269→   269→  useEffect(() => {
   270→   270→    if (!depto) return;
   271→   271→    setSt((s) => {
   272→   272→      const next = acumularNomina(s, depto);
   273→   273→      saveEmpleos(next);
   274→   274→      return next;
   275→   275→    });
   276→   276→    const t = setInterval(() => setTick((x) => x + 1), 1000);
   277→   277→    return () => clearInterval(t);
   278→   278→  }, [depto]);
   279→   279→
   280→   280→  const sueldoMin = depto ? sueldoPorMin(depto, st.xpLaboral) : 0;
   281→   281→  const rango = rangoDe(st.xpLaboral);
   282→   282→  const sueldoHora = sueldoMin * 60;
   283→   283→
   284→   284→  // proyección visual: lo que llevaría la nómina si siguiera corriendo ahora mismo
   285→   285→  const nominaViva = useMemo(() => {
   286→   286→    void tick;
   287→   287→    if (!depto || !st.lastAccrueTs) return st.nominaPendiente;
   288→   288→    const mins = Math.max(0, Math.floor((Date.now() - st.lastAccrueTs) / 60000));
   289→   289→    const yaMin = st.nominaPendiente / sueldoMin || 0;
   290→   290→    const pagables = Math.min(mins, Math.max(0, 360 - yaMin));
   291→   291→    return st.nominaPendiente + pagables * sueldoMin;
   292→   292→  }, [tick, depto, st.lastAccrueTs, st.nominaPendiente, sueldoMin]);
   293→   293→
   294→   294→  const actualizar = useCallback((next: EmpleosState) => {
   295→   295→    saveEmpleos(next);
   296→   296→    setSt(next);
   297→   297→  }, []);
   298→   298→
   299→   299→  const cobrar = () => {
   300→   300→    if (!depto) return;
   301→   301→    let s = acumularNomina(stRef.current, depto);
   302→   302→    const monto = Math.round(s.nominaPendiente);
   303→   303→    if (monto <= 0) {
   304→   304→      toast.info("La nómina está vacía: vuelve más tarde, sigue corriendo.");
   305→   305→      return;
   306→   306→    }
   307→   307→    s = { ...s, nominaPendiente: 0, cobradoTotal: s.cobradoTotal + monto };
   308→   308→    actualizar(s);
   309→   309→    addCoins(monto, `NÓMINA · ${depto.nombre}`);
   310→   310→    sfx.coin();
   311→   311→    toast.success(`Nómina cobrada: +${monto}ⓒ — ${depto.nombre}`);
   312→   312→  };
   313→   313→
   314→   314→  const firmar = () => {
   315→   315→    if (!oferta) return;
   316→   316→    const s = loadEmpleos();
   317→   317→    actualizar({
   318→   318→      ...s,
   319→   319→      depto: oferta.depto.id,
   320→   320→      entrevistas: s.entrevistas + 1,
   321→   321→      nominaPendiente: 0,
   322→   322→      lastAccrueTs: Date.now(),
   323→   323→    });
   324→   324→    setOferta(null);
   325→   325→    sfx.unlock();
   326→   326→    toast.success(`Contrato firmado: ${oferta.depto.nombre} · ${oferta.depto.sueldoBase.toFixed(1)}ⓒ/min`);
   327→   327→  };
   328→   328→
   329→   329→  const iniciarTurno = () => {
   330→   330→    if (!depto) return;
   331→   331→    if (Date.now() - st.lastShiftTs < COOLDOWN_TURNO_MS) {
   332→   332→      toast.error("Aún estás de guardia pasiva: el próximo turno abre en unos minutos.");
   333→   333→      return;
   334→   334→    }
   335→   335→    setResumenTurno(null);
   336→   336→    setTareaAbierta(true);
   337→   337→    sfx.click();
   338→   338→  };
   339→   339→
   340→   340→  const finTurno = (aciertos: number) => {
   341→   341→    if (!depto) return;
   342→   342→    const dia = hoyKey();
   343→   343→    const n = st.diaActual === dia ? st.shiftsHoy : 0;
   344→   344→    const pago = pagoTurno(depto, st.xpLaboral, aciertos, 3);
   345→   345→    let s: EmpleosState = {
   346→   346→      ...st,
   347→   347→      lastShiftTs: Date.now(),
   348→   348→      shiftsHoy: st.diaActual === dia ? st.shiftsHoy + 1 : 1,
   349→   349→      diaActual: dia,
   350→   350→      turnosTotales: st.turnosTotales + 1,
   351→   351→      aciertosTotales: st.aciertosTotales + aciertos,
   352→   352→      xpLaboral: st.xpLaboral + pago.xpGanada,
   353→   353→    };
