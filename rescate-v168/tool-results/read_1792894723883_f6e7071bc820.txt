    74→// ---------- ÓRBITA DE PALABRAS: esfera fibonacci en CSS 3D ----------
    75→function OrbitaPalabras({ palabras, acento }: { palabras: string[]; acento: string }) {
    76→  const reduced = useReducedMotion();
    77→  const dragRef = useRef<HTMLDivElement>(null);
    78→  const zonaRef = useRef<HTMLDivElement>(null);
    79→  const pose = useRef({ rx: -8, ry: 0, activo: false, x: 0, y: 0 });
    80→
    81→  // esfera fibonacci: reparte las palabras sin agruparse en los polos
    82→  const nodos = palabras.map((palabra, i) => {
    83→    const N = palabras.length;
    84→    const phi = Math.acos(-1 + (2 * i + 1) / N);
    85→    const theta = Math.sqrt(N * Math.PI) * phi;
    86→    return {
    87→      palabra,
    88→      x: Math.cos(theta) * Math.sin(phi),
    89→      y: Math.sin(phi),
    90→      z: Math.sin(theta) * Math.sin(phi),
    91→    };
    92→  });
    93→  const R = 118;
    94→
    95→  const aplicaPose = () => {
    96→    if (dragRef.current) {
    97→      dragRef.current.style.transform = `rotateX(${pose.current.rx.toFixed(1)}deg) rotateY(${pose.current.ry.toFixed(1)}deg)`;
    98→    }
    99→  };
   100→  useEffect(aplicaPose, []);
   101→
   102→  const onDown = (e: React.PointerEvent) => {
   103→    pose.current.activo = true;
   104→    pose.current.x = e.clientX;
   105→    pose.current.y = e.clientY;
   106→    (e.target as HTMLElement).setPointerCapture?.(e.pointerId);
   107→  };
   108→  const onMove = (e: React.PointerEvent) => {
   109→    if (!pose.current.activo) return;
   110→    pose.current.ry += (e.clientX - pose.current.x) * 0.45;
   111→    pose.current.rx -= (e.clientY - pose.current.y) * 0.35;
   112→    pose.current.rx = Math.max(-80, Math.min(80, pose.current.rx));
   113→    pose.current.x = e.clientX;
   114→    pose.current.y = e.clientY;
   115→    aplicaPose();
   116→  };
   117→  const onUp = () => { pose.current.activo = false; };
   118→
   119→  return (
   120→    <div
   121→      ref={zonaRef}
   122→      className="relative h-[340px] overflow-hidden rounded-md border bg-black/40"
   123→      style={{ borderColor: `${acento}47`, perspective: "820px", touchAction: "none", cursor: pose.current.activo ? "grabbing" : "grab" }}
   124→      onPointerDown={onDown}
   125→      onPointerMove={onMove}
   126→      onPointerUp={onUp}
   127→      onPointerLeave={onUp}
   128→    >
   129→      <div className="absolute inset-0" style={{ background: `radial-gradient(ellipse at 50% 62%, ${acento}14, transparent 62%)` }} />
   130→      <div ref={dragRef} className="absolute inset-0" style={{ transformStyle: "preserve-3d" }}>
   131→        <div className={cn("absolute left-1/2 top-1/2 -mt-4", !reduced && "v113-orbita")} style={{ transformStyle: "preserve-3d" }}>
   132→          {nodos.map((n, i) => (
   133→            <span
   134→              key={i}
   135→              className="v113-orbita-palabra absolute left-0 top-0 -translate-x-1/2 -translate-y-1/2 whitespace-nowrap font-mono text-[11px] font-bold uppercase tracking-widest"
   136→              style={{
   137→                color: i % 3 === 0 ? acento : "#E8EAF2",
   138→                opacity: 0.42 + n.z * 0.58,
   139→                transform: `translateZ(${(n.z * R).toFixed(1)}px) translateX(${(n.x * R).toFixed(1)}px) translateY(${(n.y * R * 0.9).toFixed(1)}px)`,
   140→              }}
   141→            >
   142→              {n.palabra}
   143→            </span>
   144→          ))}
   145→        </div>
   146→      </div>
   147→      {/* núcleo pulsante de la esfera */}
   148→      <div className="pointer-events-none absolute left-1/2 top-1/2 h-2 w-2 -translate-x-1/2 -translate-y-1/2 rounded-full v110-latido" style={{ background: acento, boxShadow: `0 0 16px ${acento}` }} />
   149→      <SelloVivo acento={acento} etiqueta="ÓRBITA" className="absolute bottom-2 right-2 z-10" />
   150→    </div>
   151→  );
   152→}

   154→// ---------- LLUVIA DE LETRAS: canvas con TU alfabeto ----------
   155→function LluviaLetras({ texto, acento }: { texto: string; acento: string }) {
   156→  const canvasRef = useRef<HTMLCanvasElement>(null);
   157→  const reduced = useReducedMotion();
   158→
   159→  useEffect(() => {
   160→    const canvas = canvasRef.current;
   161→    if (!canvas) return;
   162→    const ctx = canvas.getContext("2d");
   163→    if (!ctx) return;
   164→
   165→    const dpr = Math.min(1.5, window.devicePixelRatio || 1);
   166→    const W = canvas.clientWidth;
   167→    const H = canvas.clientHeight;
   168→    canvas.width = Math.round(W * dpr);
   169→    canvas.height = Math.round(H * dpr);
   170→    ctx.scale(dpr, dpr);
   171→
   172→    // el alfabeto ES TUYO: los glifos salen de tu texto (default VANGUARD)
   173→    const glifos = (Array.from(new Set(Array.from(texto.toUpperCase()))).filter((c) => c !== " ").slice(0, 22).join("") || "VANGUARD").split("");
   174→    const fs = 15;
   175→    const cols = Math.max(6, Math.floor(W / fs));
   176→    const gotas = Array.from({ length: cols }, () => Math.floor(Math.random() * -60));
   177→
   178→    const pinta = () => {
   179→      ctx.fillStyle = "rgba(2,4,10,0.16)";
   180→      ctx.fillRect(0, 0, W, H);
   181→      ctx.font = `bold ${fs}px ui-monospace, monospace`;
   182→      for (let c = 0; c < cols; c++) {
   183→        const ch = glifos[Math.floor(Math.random() * glifos.length)];
   184→        const y = gotas[c] * fs;
   185→        // cabeza luminosa + estela que se apaga sola
   186→        ctx.fillStyle = "#FFFFFF";
   187→        ctx.shadowColor = acento;
   188→        ctx.shadowBlur = 9;
   189→        ctx.fillText(ch, c * fs + 2, y);
   190→        ctx.shadowBlur = 0;
   191→        ctx.fillStyle = acento;
   192→        ctx.globalAlpha = 0.7;
   193→        const ch2 = glifos[Math.floor(Math.random() * glifos.length)];
   194→        ctx.fillText(ch2, c * fs + 2, y - fs);
   195→        ctx.globalAlpha = 1;
   196→        if (y > H && Math.random() > 0.975) gotas[c] = Math.floor(Math.random() * -30);
   197→        else gotas[c]++;
   198→      }
   199→    };
   200→
   201→    if (reduced) {
   202→      // reduced-motion: un fotograma estático (la lluvia no cae)
   203→      ctx.fillStyle = "rgba(2,4,10,0.85)";
   204→      ctx.fillRect(0, 0, W, H);
   205→      ctx.font = `bold ${fs}px ui-monospace, monospace`;
   206→      ctx.fillStyle = acento;
   207→      for (let c = 0; c < cols; c++) {
   208→        ctx.fillText(glifos[c % glifos.length], c * fs + 2, (Math.abs(gotas[c]) % (H / fs)) * fs);
   209→      }
   210→      return;
   211→    }
   212→
   213→    let raf = 0;
   214→    let frame = 0;
   215→    const loop = () => {
   216→      frame++;
   217→      if (frame % 2 === 0) pinta(); // ~30 fps — suave y barato
   218→      raf = requestAnimationFrame(loop);
   219→    };
   220→    raf = requestAnimationFrame(loop);
   221→    return () => cancelAnimationFrame(raf);
   222→  }, [texto, acento, reduced]);
   223→
   224→  return (
   225→    <div className="relative h-[210px] overflow-hidden rounded-md border bg-black/60" style={{ borderColor: `${acento}47` }}>
   226→      <canvas ref={canvasRef} className="absolute inset-0 h-full w-full" />
   227→      <div className="pointer-events-none absolute inset-0 bg-[radial-gradient(ellipse_at_center,transparent_55%,rgba(0,0,0,0.5)_100%)]" />
   228→    </div>
   229→  );
   230→}
